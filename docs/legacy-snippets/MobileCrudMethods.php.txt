private function updateMobileRecord(
    Request $request,
    array $tables,
    int $id,
    array $allowedColumns
): JsonResponse {
    $this->authoriseMobileWrite($request);

    $table = $this->requireTable($tables, 'Record');

    $existing = $this->scopedQuery($table, $this->propertyId($request))
        ->where('id', $id)
        ->first();

    abort_if(! $existing, 404, 'Record not found.');

    $payload = [];

    foreach ($allowedColumns as $column) {
        if (
            Schema::hasColumn($table, $column)
            && $request->exists($column)
        ) {
            $payload[$column] = $request->input($column);
        }
    }

    if (Schema::hasColumn($table, 'updated_at')) {
        $payload['updated_at'] = now();
    }

    if ($payload !== []) {
        DB::table($table)
            ->where('id', $id)
            ->update($payload);
    }

    $fresh = $this->scopedQuery($table, $this->propertyId($request))
        ->where('id', $id)
        ->first();

    return response()->json([
        'message' => 'Record updated successfully.',
        'data' => (array) $fresh,
    ]);
}

private function deleteMobileRecord(
    Request $request,
    array $tables,
    int $id
): JsonResponse {
    $this->authoriseMobileWrite($request);

    $table = $this->requireTable($tables, 'Record');

    $existing = $this->scopedQuery($table, $this->propertyId($request))
        ->where('id', $id)
        ->first();

    abort_if(! $existing, 404, 'Record not found.');

    /*
    | Prefer soft deletion when the table supports it.
    */
    if (Schema::hasColumn($table, 'deleted_at')) {
        DB::table($table)
            ->where('id', $id)
            ->update([
                'deleted_at' => now(),
                'updated_at' => Schema::hasColumn($table, 'updated_at')
                    ? now()
                    : DB::raw('updated_at'),
            ]);
    } else {
        DB::table($table)->where('id', $id)->delete();
    }

    return response()->json([
        'message' => 'Record deleted successfully.',
    ]);
}

private function authoriseMobileWrite(Request $request): void
{
    $allowed = [
        'super_admin',
        'admin',
        'hotel_owner',
        'general_manager',
        'branch_manager',
    ];

    abort_unless(
        in_array(strtolower((string) $request->user()->role), $allowed, true),
        403,
        'You do not have permission to edit or delete this record.'
    );
}
